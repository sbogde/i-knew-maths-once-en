\noindent\underline{Theorem}: \ Let $G$ be a group such that $\exists\,a,b\in G$ with the properties

\begin{enumerate}
\item[1)] $\langle a,b\rangle = G$
\item[2)] ord $a=n$, $n\geq 3$, ord $b=2$ and $ba=a^{n-1}b$
\end{enumerate}

\noindent Then $G\cong D_n$ and ord $G=2n$

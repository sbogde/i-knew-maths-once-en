\noindent\underline{\textbf{The Fundamental Theorem for Group Morphisms.}}

\noindent\underline{\textbf{(The Factorization Theorem for Group Morphisms)}}

\bigskip
\noindent $f:A\to B$ \qquad $R_f = \{(x,y)\in A\times A \mid f(x)=f(y)\}$.

\noindent $R_f$ is an equivalence relation.

\bigskip
\noindent Let $R\subseteq A\times A$ \qquad $\dfrac{A}{R} = \{\hat a \mid a\in A\}$ \qquad $\hat a = \{x\in A \mid xRa\}$.

\[
\begin{tikzcd}
A \arrow[r,"\pi_R"] \arrow[d,"f"'] & A/R \arrow[dl, dashed,"f^*"] \\
B &
\end{tikzcd}
\]

\noindent $\pi_R$ surjective, \quad $\pi_R(a) = \hat a$.

\bigskip
\noindent $\exists\,f^*: \dfrac{A}{R} \to B$ such that $f^*\circ\pi_R = f$ $\Leftrightarrow$
\[
\left\{\begin{aligned}
&R\subseteq R_f.\\
&f^*\text{ surjective} \Leftrightarrow f \text{ surjective}\\
&f^*\text{ injective} \Leftrightarrow R=R_f\\
&f \text{ bijective} \Leftrightarrow \text{-- -- --}
\end{aligned}\right.
\]

\noindent $f^*(\hat a) = f(a)$.

\bigskip
\noindent Let $f:G\to G'$ be a group morphism, $N\trianglelefteq G$, and $\pi_N(a) = \hat a = aN$

\[
\pi_N: G \to \dfrac{G}{N}.
\]

\begin{enumerate}
\item[(1)] $\exists\,f^*: \dfrac{G}{N}\to G'$ a morphism such that $f^*\circ\pi_N = f \ \Leftrightarrow \ N\subseteq\ker f$.
\item[(2)] $f^*$ injective $\Leftrightarrow N=\ker f$.
\item[(3)] $f^*$ surjective $\Leftrightarrow f$ surjective.
\item[(4)] $f^*$ an isomorphism $\Leftrightarrow N=\ker f$, $f$ surjective.
\end{enumerate}

\[
\begin{tikzcd}
G \arrow[r,"f"] \arrow[d,"\pi_N"'] & G' \\
G/N \arrow[ur, dashed,"f^*"'] &
\end{tikzcd}
\]

\noindent $f^*(\hat a) = f(a)$.

\bigskip
\noindent $R_N^L = R_N^R = R_N \subseteq G\times G$ \ for $x\equiv x' \pmod N$, $y\equiv y' \pmod N \ \Rightarrow \ xy\equiv x'y' \pmod N$

\bigskip
\noindent $R_f$ is an equivalence relation compatible with $\cdot$ on $G$.

\[
\left.\begin{aligned}
xR_fx_1 &\Leftrightarrow f(x)=f(x_1)\\
yR_fy_1 &\Leftrightarrow f(y)=f(y_1)
\end{aligned}\right\} \ \Rightarrow \ xyR_fx_1y_1 \Leftrightarrow f(xy)=f(x_1y_1).
\]

\noindent\underline{Subgroup generated by a set of elements of a group}

\noindent\underline{The commutator (derived) subgroup of a group}

\noindent\underline{The dihedral group $\Delta_n$, $n\geq 3$}

\bigskip
\bigskip
\noindent Let $(G,\cdot,e)$ be a group, $M\subseteq G$, $M\neq\varnothing$

\noindent\underline{Problem}: \ Describe the smallest subgroup of $G$

\noindent that contains $M$.

\bigskip
\noindent\underline{Prop.} \ If $H_i$, $i\in I$ is a family of subgroups of $G$ and

\[
H=\bigcap_{i\in I} H_i \quad \text{then} \quad H\leq G.
\]

\bigskip
\noindent\underline{Proof} \ TRIVIAL!

\noindent $H\neq\varnothing$ \quad $e\in H_i$ $(\forall)\,i \Rightarrow e\in H$

\noindent If $x,y\in H$ and $\left.\begin{aligned}x,y&\in H_i\ (\forall)\,i \Rightarrow xy\in H_i\ (\forall)\,i\\ x^{-1}&\in H_i\ (\forall)\,i\end{aligned}\right\} \Rightarrow xy,x^{-1}\in H$

\bigskip
\noindent\underline{Remark.} \ (1) \ If $H\leq G$ and $M\subseteq H \Rightarrow \langle M\rangle \leq H$ \ where

\[
\langle M\rangle = \{x_1^{\varepsilon_1}x_2^{\varepsilon_2}\cdots x_m^{\varepsilon_m} \mid m\geq 1,\ x_i\in M,\ \varepsilon_i=\pm 1\ (\forall)\,i\}
\]

\noindent (2) $\langle M\rangle \leq G$

\noindent (3) $M\subseteq \langle M\rangle$ \quad (the elements of $M$ are generators; $\langle M\rangle$ is the subgroup of $G$ generated by $M$)

\noindent (4) $\displaystyle\bigcap_{M\subseteq H\subseteq G} H$ = the smallest subgroup of $G$ that contains $M$ \quad (to be proved)

\bigskip
\noindent $\langle\varnothing\rangle := \langle\{e\}\rangle$

\bigskip
\noindent If $M=\{a_1,\ldots,a_n\}$ then $\langle\{a_1,\ldots,a_n\}\rangle = \langle a_1,\ldots,a_n\rangle$

\bigskip
\noindent\underline{Def.} \ Let $G$ be a group. We say $G$ is \underline{finitely generated}

\noindent if $\exists\,a_1,\ldots,a_m\in G$ such that $G=\langle a_1,\ldots,a_m\rangle$

\bigskip
\noindent $G$ is \underline{cyclic} if $\exists\,a\in G$ such that $\langle a\rangle=G$

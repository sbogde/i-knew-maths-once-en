\bigskip
\noindent\underline{Calculating the number of orbits of a finite group}

\noindent\underline{acting on a finite set}

\medskip
\noindent Let $G$ be a group, $|G|=m$, and $A$ a finite $G$-set, $|A|=n$.

\medskip
\noindent If $\{z_1,\ldots,z_q\}\subseteq A$ is a transversal for the
 equivalence relation $\sim_G$ on $A$, then
\[
A=\operatorname{orb}(z_1)\cup\operatorname{orb}(z_2)
\cup\cdots\cup\operatorname{orb}(z_q).
\]

\noindent For $a\in G$, $\operatorname{Fix}a=\{x\in A\mid ax=x\}\subseteq A$.

\medskip
\noindent\fbox{Burnside's theorem}
\qquad
\(\displaystyle q=\frac{\sum_{a\in G}|\operatorname{Fix}a|}{|G|}.\)

\bigskip
\noindent $G=S_n$,\qquad $A=\{1,2,\ldots,n\}$,
\[
S_n\times A\longrightarrow A,\qquad (\sigma,i)\longmapsto\sigma(i).
\]

\noindent Then $q=1$,\quad
$\operatorname{Fix}\sigma=\{i\in A\mid\sigma(i)=i\}$.

\[
\sum_{\sigma\in S_n}|\operatorname{Fix}\sigma|=n!.
\]

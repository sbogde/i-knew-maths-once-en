\bigskip
\noindent\fbox{11}\quad $|M|=m$, $|N|=n$.

\noindent Define
\[
 s(m,n)=\bigl|\{f:M\longrightarrow N\mid f\text{ is surjective}\}\bigr|.
\]
\noindent Compute $s(m,n)$. \hfill $s(2,1)=1$.

\noindent\underline{Solution.}
\[
 s(m,n)=0\qquad\text{when }n>m.
\]
\noindent If $n=m$, then
\[
 s(m,n)=n!.
\]
\noindent If $n<m$, then
\[
 |\operatorname{Hom}(M,N)|=n^m.
\]
\noindent Let $P=\{f:M\longrightarrow N\mid f\text{ is not surjective}\}$.
Hence
\[
 s(m,n)=n^m-|P|.
\]
\noindent Write
\[
 M=\{a_1,a_2,\ldots,a_m\},\qquad
 N=\{b_1,b_2,\ldots,b_n\}.
\]
\noindent For $i=1,\ldots,n$, let
\[
 A_i'=\{f:M\longrightarrow N\mid b_i\notin\operatorname{Im}f\}.
\]
\noindent Then
\[
 |P|=\bigl|A_1'\cup A_2'\cup\cdots\cup A_n'\bigr|,
\]
\[
 |A_i'|=(n-1)^m\quad(\forall i),\qquad
 |A_i'\cap A_j'|=(n-2)^m.
\]
\noindent By inclusion-exclusion,
\[
\begin{aligned}
 |P|={}&\binom{n}{1}(n-1)^m
       -\binom{n}{2}(n-2)^m\\
      &+\binom{n}{3}(n-3)^m-\cdots
       +(-1)^{n-1}0.
\end{aligned}
\]

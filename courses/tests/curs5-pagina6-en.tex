\noindent\underline{\textbf{The order of an element in a group.}}

\bigskip
\noindent\underline{Def}: Let $G$ be a group, $a\in G$. We say $a$ is an element

\noindent of finite order (a torsion element of the group $G$) if $\exists\, k\in\mathbb{N}^*$

\noindent such that $a^k=e$.

\bigskip
\noindent In this case $n=\min\{k \mid k\in\mathbb{N}^*, a^k=e\}$ is called the \underline{order of $a$}

\noindent denoted ord $a$

\bigskip
\noindent if $\nexists\, k\in\mathbb{N}^*$ such that $a^k=e$ \quad $a$ has \underline{infinite} order

\bigskip
\noindent\underline{Lemma 1}: Let $G$ be a group, $a\in G$.

\noindent We show the following are equivalent:

\begin{enumerate}
\item[(1)] ord$(a)<\infty$.
\item[(2)] $f:\mathbb{Z}\to G$ \quad $f(k)=a^k$ is not injective.
\end{enumerate}

\bigskip
\noindent\underline{Proof}: $(1)\Rightarrow(2)$. \ Suppose ord$(a)=r \overset{\text{def}}{=} r\in\mathbb{N}^*$.

\[
\Rightarrow a^0=a^r=e, \ r\neq 0 \ \Rightarrow \ f \text{ is not injective.}
\]

\bigskip
\noindent $(2)\Rightarrow(1)$. \ $f$ not injective $\Rightarrow \exists\, i\neq j\in\mathbb{Z}$ such that $f(i)=f(j)$.

\noindent $(i<j)$.

\[
a^i=a^j \Rightarrow a^{j-i}=e.
\]
\[
j-i=k>0
\]

\bigskip
\noindent\underline{Consequence}: ord$(G)<+\infty \Rightarrow$ ord$(a)<\infty$ $(\forall)\,a\in G$.

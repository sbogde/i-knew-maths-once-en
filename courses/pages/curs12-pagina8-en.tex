\subsection*{Lecture 12 - Page 8}

\noindent\fbox{T}
\begin{enumerate}
\item[(1)] $\mathrm{FG}[A]$ is a group.
\item[(2)] The map $\zeta_A:A\longrightarrow\mathrm{FG}[A]$
is injective.
\end{enumerate}

\medskip
\noindent For every group $(G,\cdot,e)$ and every map
$\zeta:A\longrightarrow G^*$,
\[
\exists!\,g:\mathrm{FG}[A]\longrightarrow G
\quad\text{such that}\quad g\circ\zeta_A=\zeta.
\]

\[
\begin{tikzcd}[column sep=large, row sep=large]
A \arrow[r,"\zeta_A"] \arrow[dr,"\zeta"']
  & \mathrm{FG}[A] \arrow[d,dashed,"g"]\\
& G
\end{tikzcd}
\]

\medskip
\noindent $A\subseteq A\cup A'$,
\[
A\cup A'\xrightarrow{\ \zeta_{A\cup A'}\ }(A\cup A')^*
\xrightarrow{\ \rho\ }
\frac{(A\cup A')^*}{R^*}
=\langle\overline{A}\cup\overline{A'}\rangle.
\]

\[
\begin{tikzcd}[column sep=large, row sep=large]
A\cup A' \arrow[r,"\zeta_{A\cup A'}"] \arrow[dr,"\zeta_1"']
  & (A\cup A')^* \arrow[r,"\rho"] \arrow[d,dashed,"f"]
  & \mathrm{FG}[A] \arrow[dl,dashed,"g"]\\
& G &
\end{tikzcd}
\]

\noindent
\[
\zeta_1(x)=\zeta(x)\quad(\forall)\,x\in A,
\qquad
\zeta_1(x')=\zeta(x)^{-1}\quad(\forall)\,x'\in A'.
\]

\noindent $f\circ\zeta_{A\cup A'}=\zeta_1$,
\qquad $R^*\subseteq R_f$.

\medskip
\noindent $(xx',1)\in R\subseteq R_f$.
\[
f(xx')=f(x)f(x')=e=f(1),
\]
\noindent where $f(x)=\zeta_1(x)=\zeta(x)$ and
$f(x')=\zeta_1(x')=\zeta(x)^{-1}$.

\bigskip
\noindent\underline{\textit{``Word problem''}}
\[
R\subseteq A^*\times A^*,\qquad
R^*\subseteq A^*\times A^*,\qquad
u,v\in A^*,\quad u\,R^*\,v.
\]

\noindent Is there an algorithm such that $v$ is obtained from $u$
in a finite number of steps?

\medskip

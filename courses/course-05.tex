\noindent\underline{\textbf{Lagrange's Theorem}}

\bigskip
\noindent Let $(G,\cdot)$ be a group. $H\leq G$.

\noindent If $x,y\in G$ \quad $x\equiv_L y \pmod H \overset{\text{def}}{\Longleftrightarrow} x^{-1}y\in H$.

\[
R_H^L = \{(x,y)\in G\times G \mid x^{-1}y\in H\} \subseteq G\times G.
\]

\noindent $R_H^L$ is an equivalence relation on $G$, compatible on the left with $\cdot$.

\[
xR_H^L y \Rightarrow zxR_H^L zy
\]

\noindent For $a\in G$, \ $[a]_{R_H^L} = aH = \{ah \mid h\in H\}$ = the left coset

\bigskip
\noindent $x\equiv_R y \pmod H \overset{\text{def}}{\Longleftrightarrow} xy^{-1}\in H$.

\[
R_H^R = \{(x,y)\in G\times G \mid xy^{-1}\in H\} = \text{an equivalence relation compatible on the right with } \cdot.
\]

\[
xR_H^R y \Rightarrow xzR_H^R yz.
\]

\[
[a]_{R_H^R} = Ha = \{ha \mid h\in H\}.
\]

\bigskip
\noindent Hence $T=\{a_i\}_{i\in I}$ = a left transversal of $H$ in $G$

\noindent = a complete system of representatives

\begin{enumerate}
\item[1)] $i\neq j \Rightarrow a_i \not\equiv_L a_j \pmod H$
\item[2)] $(\forall)\, x\in G$ \ $\exists\, i\in I$ such that $x\equiv_L a_i \pmod H$
\end{enumerate}

\bigskip
\noindent\underline{Notation.} $T^{-1} = \{a_i^{-1}\}_{i\in I}$

\noindent $T$ a left transversal $\Rightarrow$ $T^{-1}$ is a right transversal.

\[
T \xrightarrow{\ } T^{-1}, \qquad a_i \mapsto a_i^{-1} \text{ is a bijection} \ \Rightarrow \ \text{card}\, T = \text{card}\, T^{-1}
\]

\noindent = the index of $H$ in $G \overset{\text{not}}{=} [G:H]$

\noindent ord $G$ = card $G$ = $|G|$.

\bigskip
\noindent\fbox{Lagrange's Theorem} \quad If $(G,\cdot)$ is a finite group and $H$ is a subgroup of $G$,

\noindent then:

\[
\boxed{|G| = |H|\cdot [G:H]}
\]

\noindent or, equivalently: \quad $\big($ord $G$ = ord $H\cdot [G:H]\big)$

\bigskip
\noindent\underline{Application:} ord $S_n = n!$

\[
S_n = \{\sigma : \{1,2,\ldots,n\}\to\{1,2,\ldots,n\} \mid \sigma \text{ is bijective}\}.
\]

% \noindent [Notation:]\footnote{the connecting fragment here is very compressed in the original.}

\[
\sigma = \begin{pmatrix} 1 & 2 & \cdots & n \\ \sigma(1) & \sigma(2) & \cdots & \sigma(n) \end{pmatrix}
\]

\noindent Induction on $n$.

\bigskip
\noindent $n=1$. \quad ord $S_1=1$.

\bigskip
\noindent Suppose ord $S_{n-1}=(n-1)!$.

\noindent\underline{ord $S_n=n!$}

\bigskip
\noindent Let $\sigma_i\in S_n$ be such that $\sigma_i(n)=i$, $(\forall)\, i=\overline{1,n}$.

\bigskip
\noindent $H=\{\sigma\in S_n \mid \sigma(n)=n\}$. \quad 1) \underline{$H\leq S_n$}, [ord $\underline{H=(n-1)!}$]

\bigskip
\noindent (3) $\{\sigma_1,\sigma_2,\ldots,\sigma_n\}$ is a left (and right) transversal of $H$ in $S_n$.

\bigskip
\noindent (1) $\sigma,\bar\alpha\in H$, $H\neq\varnothing$,

\[
\sigma(\bar\alpha(n))=\sigma(n)=n \ \Rightarrow \ \sigma\circ\bar\alpha\in H \ \Rightarrow \ H \text{ is a stable subset.}
\]

\noindent $\sigma\in H \Rightarrow \sigma(n)=n \ \Rightarrow \ \sigma^{-1}(\sigma(n))=\sigma^{-1}(n) \ \Rightarrow \ n=\sigma^{-1}(n) \ \Rightarrow$

\[
\Rightarrow \ \sigma^{-1}\in H \ \Rightarrow \ \underline{H\leq S_n}.
\]

\bigskip
\noindent (2) \quad $S_{n-1} \xrightarrow{\ f\ } S_n$

\[
\alpha = \begin{pmatrix} 1 & 2 & \cdots & n-1 \\ \alpha(1) & \alpha(2) & \cdots & \alpha(n-1) \end{pmatrix}
\]
\[
f(\alpha) = \begin{pmatrix} 1 & 2 & \cdots & n-1 & n \\ \alpha(1) & \alpha(2) & \cdots & \alpha(n-1) & n \end{pmatrix} \in H.
\]

\noindent $f$ is a bijection between $S_{n-1}$ and $H$ -- evident.

\[
\Rightarrow S_{n-1}\cong H \overset{\text{induction hyp.}}{\Longrightarrow} \text{ord } H=(n-1)!.
\]

\bigskip
\noindent (3) \quad $\big[i\neq j \Rightarrow \sigma_i \not\equiv_L \sigma_j \pmod H\big]$.

\noindent $\big[(\forall)\, \sigma\in S_n \ \exists!\, i \text{ such that } \sigma\equiv_L \sigma_i \pmod H\big]$.

\bigskip
\noindent Suppose $\sigma_i\equiv_L\sigma_j \pmod H \Rightarrow \sigma_i^{-1}\circ\sigma_j\in H$

\noindent $\Rightarrow \sigma_i^{-1}(\sigma_j(n))=n. \Rightarrow \sigma_j(n)=\sigma_i(n) \Rightarrow j=i$

\bigskip
\noindent Let $i=\sigma(n)$. We claim $\sigma\equiv_L\sigma_i \pmod H \Leftrightarrow \sigma^{-1}\sigma_i\in H$.

\[
\sigma^{-1}(\sigma_i(n))=n \Leftrightarrow \underbrace{\sigma_i(n)}_{i} = \underbrace{\sigma(n)}_{i} \quad \text{true.}
\]

\bigskip
\noindent\underline{Lagrange} $\Rightarrow$ ord $S_n$ = ord $H\cdot[S_n:H]$ = ord $H\cdot$ord$\{\sigma_1,\ldots,\sigma_n\}$

\[
= (n-1)!\cdot n.
\]

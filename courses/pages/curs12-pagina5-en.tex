\subsection*{Lecture 12 - Page 5}

\noindent\underline{Application 1}\quad
``Free commutative monoid generated by a set''.

\[
R=\bigl\{(xy,yx)\mid x,y\in A^*\bigr\},\qquad R^{-1}=R.
\]

\medskip
\noindent\underline{Application 2}\quad Free group generated by a set.

\bigskip
\noindent\textbf{1.} If $u,v\in A^*$, we say that $v$ is obtained
 from $u$ by an \underline{$R$-substitution} if
\[
\exists\,(w,w')\in R\cup R^{-1}\cup\{(1,1)\},
\qquad \exists\,p,q\in A^*
\]
\noindent such that
\[
u=pwq,\qquad v=pw'q.
\]

\noindent (That is, the two words contain the elements of a pair as subwords;
 they have the same initial and final parts.)

\bigskip
\noindent $S_R\subseteq A^*\times A^*$ is a binary relation,
\[
u\,S_R\,v\quad\Longleftrightarrow\quad
v\text{ is obtained from }u\text{ by an }R\text{-substitution}.
\]

\noindent $S_R$ is reflexive, symmetric, and compatible on the left
 and on the right with the operation ``$\cdot$'' on $A^*$.
\[
u\,S_R\,v\ \overset{?}{\Longrightarrow}\
\begin{cases}
zu\,S_R\,zv, & (\forall)\,z\in A^*,\\
uz\,S_R\,vz, & (\forall)\,z\in A^*.
\end{cases}
\]
\noindent $zu=zpwq$ and $zv=zpw'q$.

\medskip
\noindent $u\,S_R\,u\;?$ \quad $(\forall)\,u\in A^*$.
\[
u=u\cdot1\cdot1,\qquad u=u\cdot1\cdot1
\quad\bigl(w=w'=1\bigr).
\]

\noindent $u\,S_R\,v\ \overset{?}{\Longrightarrow}\ v\,S_R\,u$.

\bigskip
\noindent\underline{Remark.}\quad $R\subseteq S_R$.
\[
(w,w')\in R,\qquad p=q=1
\quad\Longrightarrow\quad (w,w')\in S_R.
\]

\bigskip
\noindent $R^*\subseteq A^*\times A^*$, where for $u,v\in A^*$,
\[
(u,v)\in R^*\ \Longleftrightarrow\
\begin{gathered}
\exists\,u_1,\ldots,u_n\in A^*,\quad n\text{ finite},\\
u=u_1,\quad v=u_n,\quad
u_i\,S_R\,u_{i+1}\quad (\forall)\,i=1,\ldots,n-1.
\end{gathered}
\]
% The word above n is crossed out and replaced by 'finite' in the original.

\bigskip
\noindent\underline{Ex.}\quad $A=\{x,y,z\}$,
\[
R=\bigl\{(xy,yx),(yz,zy),(zx,xz)\bigr\}.
\]
\[
u=xxzyyx,\qquad v=yzxxyx.
\]

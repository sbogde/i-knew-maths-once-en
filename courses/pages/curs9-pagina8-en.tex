\section*{Lecture 9 - Page 8}

\noindent\underline{Proof.}

\bigskip
\noindent (1) If $a,b\in\operatorname{Stab}_G(x)$, then
\[
\left.\begin{aligned}
 ax&=x,\\
 bx&=x
\end{aligned}\right\}
\quad\Longrightarrow\quad (ab)x=a(bx)=ax=x.
\]
\[
a^{-1}x=a^{-1}(ax)=(a^{-1}a)x=ex=x.
\]
\[
\Longrightarrow\quad \operatorname{Stab}_G(x)\leq G.
\]

\bigskip
\noindent (3) $\Longleftarrow$ (2).

\bigskip
\noindent (2) $\operatorname{Orb}(x)=Gx=\{ax\mid a\in G\}$.

\bigskip
\noindent Let $a,b\in G$. Under what conditions is $ax=bx$?

\bigskip
\noindent Now $ax=bx\ \Longleftrightarrow\ a^{-1}b\in\operatorname{Stab}_G(x)\leq G$.
\[
\big(\Longleftrightarrow\ a^{-1}bx=x\ \Longleftrightarrow\ bx=ax\big).
\]
